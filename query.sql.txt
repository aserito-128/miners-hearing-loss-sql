WorstEarCalculated AS (
  SELECT
    m.miner_id,
    CONCAT(m.first_name, ' ', m.last_name) AS worker_name,
    m.smoking_status,
    e.exam_date,
    IF(e.left_ear_loss_dB >= e.right_ear_loss_dB, e.left_ear_loss_dB, e.right_ear_loss_dB) AS worst_ear_loss_dB
  FROM miners m
  JOIN audiometric_exams e ON m.miner_id = e.miner_id  
),


YearlyProgression AS (
  SELECT
    miner_id,
    worker_name,
    smoking_status,
    exam_date,
    worst_ear_loss_dB,
    LAG(exam_date, 1) OVER (PARTITION BY miner_id ORDER BY exam_date) AS prev_exam_date,
    LAG(worst_ear_loss_dB, 1) OVER (PARTITION BY miner_id ORDER BY worst_ear_loss_dB) AS prev_worst_loss
  FROM WorstEarCalculated
)


SELECT
  miner_id,
  worker_name,
  smoking_status,
  prev_exam_date,
  exam_date AS current_exam_date,
  prev_worst_loss AS prev_exam_loss_dB,
  worst_ear_loss_dB AS current_exam_loss_dB,
  ROUND(worst_ear_loss_dB - COALESCE(prev_worst_loss, worst_ear_loss_dB), 2) AS dB_worsening_delta
FROM YearlyProgression
ORDER BY miner_id, current_exam_date;
